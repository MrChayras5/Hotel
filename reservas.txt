
-- Creación de la tabla RESERVAS
CREATE TABLE RESERVAS (
    ID_RESERVA NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    ID_HUESPED NUMBER NOT NULL,       -- Solo el ID, sin relación FK
    ID_HABITACION NUMBER NOT NULL,    -- Solo el ID, sin relación FK
    FECHA_ENTRADA DATE NOT NULL,
    FECHA_SALIDA DATE NOT NULL,
    NOCHES NUMBER NOT NULL,
    TOTAL NUMBER(10, 2) NOT NULL,     -- Para manejar decimales de dinero
    ESTADO VARCHAR2(20) NOT NULL,     -- 'CONFIRMADA', 'EN_CURSO', etc.
    FECHA_CREACION TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

ALTER TABLE RESERVAS ADD CONSTRAINT CHK_FECHAS CHECK (FECHA_SALIDA > FECHA_ENTRADA);
