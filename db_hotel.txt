Creacion de usuarios 
CREATE TABLE USUARIO (
    ID_USUARIO NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    USERNAME VARCHAR2(20) NOT NULL UNIQUE,
    PASSWORD VARCHAR2(255) NOT NULL,
    ROL VARCHAR2(10) NOT NULL,
    ACTIVO NUMBER(1) DEFAULT 1 NOT NULL,
    CONSTRAINT CK_USUARIO_ROL CHECK (ROL IN ('ADMIN', 'USER'))
);

CREATE TABLE HUESPED (
    ID_HUESPED NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    NOMBRE VARCHAR2(50) NOT NULL,
    APELLIDO VARCHAR2(50) NOT NULL,
    EMAIL VARCHAR2(100) NOT NULL UNIQUE,
    TELEFONO VARCHAR2(15) NOT NULL UNIQUE,
    DOCUMENTO VARCHAR2(20) NOT NULL,
    NACIONALIDAD VARCHAR2(50) NOT NULL
);

CREATE TABLE HABITACION (
    ID_HABITACION NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    NUMERO NUMBER NOT NULL UNIQUE,
    TIPO VARCHAR2(20) NOT NULL,
    DESCRIPCION VARCHAR2(255),
    PRECIO NUMBER(10,2) NOT NULL,
    CAPACIDAD NUMBER NOT NULL,
    ESTADO VARCHAR2(20) NOT NULL,
    CONSTRAINT CK_HABITACION_PRECIO CHECK (PRECIO > 0),
    CONSTRAINT CK_HABITACION_CAPACIDAD CHECK (CAPACIDAD >= 1),
    CONSTRAINT CK_HABITACION_ESTADO CHECK (
        ESTADO IN ('Disponible', 'Ocupada', 'Limpieza', 'Mantenimiento')
    )
);

CREATE TABLE RESERVA (
    ID_RESERVA NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    FECHA_ENTRADA DATE NOT NULL,
    FECHA_SALIDA DATE NOT NULL,
    NOCHES NUMBER NOT NULL,
    TOTAL NUMBER(10,2) NOT NULL,
    ESTADO VARCHAR2(20) NOT NULL,

    ID_HUESPED NUMBER NOT NULL,
    ID_HABITACION NUMBER NOT NULL,
    ID_USUARIO NUMBER NOT NULL,

    CONSTRAINT CK_RESERVA_FECHAS CHECK (FECHA_ENTRADA < FECHA_SALIDA),
    CONSTRAINT CK_RESERVA_TOTAL CHECK (TOTAL >= 0),
    CONSTRAINT CK_RESERVA_ESTADO CHECK (
        ESTADO IN ('Confirmada', 'En curso', 'Finalizada', 'Cancelada')
    ),

    CONSTRAINT FK_RESERVA_HUESPED
        FOREIGN KEY (ID_HUESPED) REFERENCES HUESPED(ID_HUESPED),

    CONSTRAINT FK_RESERVA_HABITACION
        FOREIGN KEY (ID_HABITACION) REFERENCES HABITACION(ID_HABITACION),

    CONSTRAINT FK_RESERVA_USUARIO
        FOREIGN KEY (ID_USUARIO) REFERENCES USUARIO(ID_USUARIO)
);
