-- Creación de la tabla USUARIOS (Para Recepcionistas y Admins)
CREATE TABLE USUARIOS (
    ID_USUARIO NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    USERNAME VARCHAR2(20) NOT NULL,       -- Entre 5 y 20 caracteres [cite: 118]
    PASSWORD VARCHAR2(255) NOT NULL,      -- Se guardará encriptada (BCrypt)
    ROLE VARCHAR2(20) NOT NULL,           -- 'ROLE_ADMIN' o 'ROLE_USER'
    ESTADO NUMBER(1) DEFAULT 1 NOT NULL   -- 1=Activo, 0=Inactivo (Lógica soft delete)
);

-- Constraint de unicidad
ALTER TABLE USUARIOS ADD CONSTRAINT UQ_USERNAME UNIQUE (USERNAME);
