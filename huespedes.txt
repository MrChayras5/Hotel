-- Creación de la tabla HUESPEDES
CREATE TABLE HUESPEDES (
    ID_HUESPED NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    NOMBRE VARCHAR2(100) NOT NULL,
    APELLIDO VARCHAR2(100) NOT NULL,
    EMAIL VARCHAR2(100) NOT NULL,
    TELEFONO VARCHAR2(15) NOT NULL,       -- Varchar es mejor para teléfonos (formato)
    DOCUMENTO VARCHAR2(20) NOT NULL,      -- INE, Pasaporte, etc.
    NACIONALIDAD VARCHAR2(50) NOT NULL
);
