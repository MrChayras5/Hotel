-- Creación de la tabla HABITACIONES
CREATE TABLE HABITACIONES (
    ID_HABITACION NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    NUMERO NUMBER NOT NULL,               -- Número de habitación (ej: 101, 205)
    TIPO VARCHAR2(50) NOT NULL,           -- 'INDIVIDUAL', 'DOBLE', 'SUITE'
    DESCRIPCION VARCHAR2(255),            -- Descripción opcional
    PRECIO NUMBER(10, 2) NOT NULL,        -- Precio por noche
    CAPACIDAD NUMBER NOT NULL,            -- Cantidad de personas (ej: 2)
    ESTADO VARCHAR2(20) DEFAULT 'DISPONIBLE' NOT NULL -- 'DISPONIBLE', 'OCUPADA', etc.
);

ALTER TABLE HABITACIONES ADD CONSTRAINT UQ_NUMERO_HAB UNIQUE (NUMERO);
ALTER TABLE HABITACIONES ADD CONSTRAINT CHK_PRECIO_POS CHECK (PRECIO > 0);
ALTER TABLE HABITACIONES ADD CONSTRAINT CHK_CAPACIDAD_POS CHECK (CAPACIDAD > 0);
